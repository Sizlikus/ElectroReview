name: Build Windows demo
on:
  workflow_dispatch:
  push:
    branches: [main, master]
permissions:
  contents: read
jobs:
  build:
    runs-on: windows-2022
    timeout-minutes: 35
    steps:
      - uses: actions/checkout@v7
      - uses: actions/setup-python@v7
        with:
          python-version: '3.12'
      - name: Install Python dependencies
        run: |
          python -m pip install -r requirements.txt "pyinstaller>=6,<7"
          if ($LASTEXITCODE -ne 0) { throw 'Python dependencies failed' }
      - name: Test review engine
        run: |
          python -m unittest discover -s tests -v
          if ($LASTEXITCODE -ne 0) { throw 'Tests failed' }
      - name: Install free Windows OCR and Russian language
        uses: msys2/setup-msys2@v2
        with:
          msystem: UCRT64
          update: true
          install: >-
            mingw-w64-ucrt-x86_64-tesseract-ocr
            mingw-w64-ucrt-x86_64-tesseract-data-rus
            mingw-w64-ucrt-x86_64-tesseract-data-eng
      - name: Locate OCR and record versions
        shell: msys2 {0}
        run: |
          echo "OCR_PREFIX=$(cygpath -w /ucrt64)" >> "$GITHUB_ENV"
          pacman -Q > msys2-packages.txt
      - name: Build and bundle
        shell: pwsh
        run: ./scripts/build-windows.ps1
      - name: Test packaged EXE from a folder with spaces and Cyrillic
        timeout-minutes: 4
        shell: pwsh
        run: |
          $testRoot = Join-Path $env:RUNNER_TEMP 'Проверка программы'
          New-Item -ItemType Directory -Force $testRoot | Out-Null
          Copy-Item 'dist/ElectroReview' $testRoot -Recurse
          $exe = Join-Path $testRoot 'ElectroReview/ElectroReview.exe'
          $report = Join-Path $env:GITHUB_WORKSPACE 'packaged-selftest.json'
          $process = Start-Process -FilePath $exe -ArgumentList @('--self-test', "`"$report`"") -Wait -PassThru
          if (Test-Path $report) { Get-Content $report -Raw }
          Write-Host "Packaged process exit: $($process.ExitCode)"
          if ($process.ExitCode -ne 0) { throw 'Packaged application self-test failed' }
          if (!(Test-Path $report)) { throw 'No self-test report' }
          $result = Get-Content $report -Raw | ConvertFrom-Json
          if ($result.status -ne 'passed') { throw 'Self-test did not pass' }
          Copy-Item $report 'dist/ElectroReview/'
      - name: Upload ready Windows folder
        uses: actions/upload-artifact@v7
        with:
          name: ElectroReview-Windows-0.2.0
          path: dist/ElectroReview/
          if-no-files-found: error
          retention-days: 30
      - name: Upload diagnostics
        if: always()
        uses: actions/upload-artifact@v7
        with:
          name: build-diagnostics
          path: |
            packaged-selftest.json
            msys2-packages.txt
          if-no-files-found: ignore
